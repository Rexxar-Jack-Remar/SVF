# Paper comparison implementations from gpu4pta 21d73b4, pestrie/src.
add_library(SvfClientMatrix STATIC matrix-ops.cc bitmap.cc obstack.cc profile_helper.cc)
add_library(SvfClientBaselines STATIC pes-querier.cc bit-querier.cc)
target_link_libraries(SvfClientBaselines PUBLIC SvfClientMatrix)
add_executable(svf-bit-index bit-indexer.cc bit-pt.cc bit-se.cc)
add_executable(svf-pes-index pes-indexer.cc pes-common.cc pes-self.cc pes-dual.cc treap.cc segtree.cc)
foreach(tool svf-bit-index svf-pes-index)
  target_link_libraries(${tool} PRIVATE SvfClientMatrix)
  set_target_properties(${tool} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
endforeach()
foreach(target SvfClientMatrix SvfClientBaselines svf-bit-index svf-pes-index)
  target_compile_options(${target} PRIVATE -w)
endforeach()
install(TARGETS svf-bit-index svf-pes-index RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
